cmake_minimum_required(VERSION 3.16)
project(variant_D LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
add_library(variant_D SHARED txxtea128.c txxtea_etm.c)
add_library(variant_D_static STATIC txxtea128.c txxtea_etm.c)
set_target_properties(variant_D_static PROPERTIES OUTPUT_NAME variant_D)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
target_link_libraries(variant_D PRIVATE OpenSSL::Crypto)
target_link_libraries(variant_D_static PUBLIC OpenSSL::Crypto)
install(TARGETS variant_D variant_D_static
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/txxtea128.h txxtea_etm.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
add_executable(bench_etm src/bench_etm.c)
target_link_libraries(bench_etm PRIVATE variant_D_static)
add_executable(demo_etm src/demo_etm.c)
target_link_libraries(demo_etm PRIVATE variant_D_static)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME correctness COMMAND ${CMAKE_COMMAND} -E env
      "CC=${CMAKE_C_COMPILER}" ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py)
  endif()
endif()
