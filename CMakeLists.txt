cmake_minimum_required(VERSION 3.16)
project(variant_E LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
add_library(variant_E SHARED txxtea128.c txxtea_etm_ctx.c)
add_library(variant_E_static STATIC txxtea128.c txxtea_etm_ctx.c)
set_target_properties(variant_E_static PROPERTIES OUTPUT_NAME variant_E)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
target_link_libraries(variant_E PRIVATE OpenSSL::Crypto)
target_link_libraries(variant_E_static PUBLIC OpenSSL::Crypto)
install(TARGETS variant_E variant_E_static
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/txxtea128.h txxtea_etm_ctx.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
add_executable(bench_etm_ctx src/bench_etm_ctx.c)
target_link_libraries(bench_etm_ctx PRIVATE variant_E_static)
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME correctness COMMAND ${CMAKE_COMMAND} -E env
      "CC=${CMAKE_C_COMPILER}" ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py)
  endif()
endif()
