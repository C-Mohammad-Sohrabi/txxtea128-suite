cmake_minimum_required(VERSION 3.16)
project(xxtea LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
add_library(xxtea SHARED src/xxtea.c)
add_library(xxtea_static STATIC src/xxtea.c)
set_target_properties(xxtea_static PROPERTIES OUTPUT_NAME xxtea)
install(TARGETS xxtea xxtea_static
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/xxtea.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
add_executable(bench src/bench.c)
target_link_libraries(bench PRIVATE xxtea_static)
add_executable(xxtea_example example/test.c example/base64.c)
target_include_directories(xxtea_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(xxtea_example PRIVATE xxtea_static)
if(BUILD_TESTING)
  add_test(NAME example COMMAND xxtea_example)
endif()
if(BUILD_TESTING)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME correctness COMMAND ${CMAKE_COMMAND} -E env
      "CC=${CMAKE_C_COMPILER}" ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py)
  endif()
endif()
